import './style.css';
import {useState} from 'react';
import {useCommerce,useAssetUrl} from '@baci/experience-sdk';
export default function Store({route,catalog}) {
 const commerce=useCommerce(), image=useAssetUrl('synthetic-device-image');
 const [cart,setCart]=useState(null),[bag,setBag]=useState(false),[message,setMessage]=useState(''),[busy,setBusy]=useState(false),[color,setColor]=useState('Sage');
 const p=catalog.products[0], count=cart?.lines.reduce((n,l)=>n+l.quantity,0)??0;
 const money=(n,c='NGN')=>new Intl.NumberFormat('en-NG',{style:'currency',currency:c,maximumFractionDigits:0}).format(n);
 async function read(){setBag(true);try{const r=await commerce.cart.readPublic();if(r.status==='ok')setCart(r.value);else setMessage(r.code??r.status);}catch{setMessage('Could not refresh your bag. Please try again.');}}
 async function add(){if(!p||busy)return;setBusy(true);try{const r=await commerce.cart.add({productId:p.id,quantity:1,...(p.requiresVariant?{variantId:p.variants[0]?.id}:{})});if(r.status==='ok'){setCart(r.value);setBag(true);setMessage('Added to your bag');}else setMessage(r.code??r.status);}catch{setMessage('Could not add this item. Please try again.');}finally{setBusy(false);}}
 async function checkout(){setBusy(true);try{const r=await commerce.checkout.prepare();if(r.status==='ok'&&r.value.ready){const opened=await commerce.checkout.open();setMessage(opened.status==='ok'?'Checkout ready':opened.code??opened.status);}else setMessage(r.code??'Add a device before checkout');}catch{setMessage('Checkout could not open. Please retry.');}finally{setBusy(false);}}
 const go=(path)=>commerce.navigation.go(path),product=()=>go('/products/'+p?.slug),isProduct=route.routeId==='product',story=route.routeId==='editorial',collection=route.routeId==='collection';
 function Device({small=false}){return <div className={'device-scene '+(small?'small':'')} aria-label="Illustration of a synthetic concept device"><div className="orbit"/><div className={'phone back '+color.toLowerCase()}><div className="cameras"><i/><i/><i/></div><span className="device-mark">o</span></div><div className="phone front"><img src={image??undefined} alt="Synthetic concept device screen"/><div className="screen-art"><span>09:41</span><strong>Make<br/>room.</strong><div className="screen-circle"/></div><div className="island"/></div><span className="art-caption">FORM / 01 — CONCEPT SERIES</span></div>}
 return <div className="store"><div className="announcement">A little less ordinary. A lot more possibility. <span>Explore the collection ↗</span></div><header><button className="wordmark" onClick={()=>go('/')} aria-label="Orbit home">orbit<span>®</span></button><nav aria-label="Main navigation"><button onClick={()=>go('/collections/all')}>The collection</button><button onClick={()=>go('/pages/story')}>Our philosophy</button></nav><button className="bag-button" onClick={read}>Bag <span>{count}</span></button></header>
 <main><section className={'hero '+(story?'story':'')}><div className="hero-copy"><p className="eyebrow">INDEPENDENT BY DESIGN / EST. 2026</p><h1>{story?<>Good design.<br/>Less noise.</>:isProduct?<>Your everyday.<br/><em>Reimagined.</em></>:collection?<>Objects for<br/><em>what’s next.</em></>:<>Life, with<br/>a little <em>orbit.</em></>}</h1><p className="intro">{story?'We believe the things you carry should earn their place. Considered details. Honest materials. Technology that leaves room for life.':'Considered tech for curious minds. Beautifully simple objects that make the everyday feel extraordinary.'}</p><button className="primary" onClick={story?()=>go('/collections/all'):isProduct?add:product}>{isProduct?(busy?'Adding…':'Add to bag — '+(p?money(p.price,p.currency):'')):story?'Find your everyday':'Meet your next everyday'} <span>↗</span></button><div className="hero-note"><span className="dot"/> Designed to stand apart. Built to fit right in.</div></div><Device/><div className="edition">THE EVERYDAY EDIT <span>01 / 03</span></div></section>
 <section className="benefits" aria-label="Store features"><span>↗ Thoughtfully selected</span><span>◎ Baci-powered checkout</span><span>↔ Made for your everyday</span><span>✳ A fresh perspective</span></section>
 <section className="collection"><div className="section-heading"><div><p className="eyebrow">FEWER THINGS. BETTER THINGS.</p><h2>{isProduct?'The details make the difference.':'Your next favourite thing.'}</h2></div><button className="text-link" onClick={()=>go('/collections/all')}>Explore the edit ↗</button></div><div className="product-layout"><button className="product-art" onClick={product} aria-label="View concept device"><span className="pill">THE CONCEPT SERIES</span><Device small/><span className="art-plus">↗</span></button><div className="product-copy"><p className="eyebrow">01 / THE EVERYDAY ESSENTIAL</p><h3>Small detail.<br/>Big possibility.</h3><p>A fresh perspective on the device you reach for most. Quiet confidence, playful colour, and room for everything that matters.</p><div className="swatches" aria-label="Illustration colour">{['Sage','Ink','Clay'].map(c=><button key={c} className={c.toLowerCase()+(color===c?' selected':'')} onClick={()=>setColor(c)} aria-label={'Preview '+c} aria-pressed={color===c}/>)}<span>{color} · illustration only</span></div><div className="product-price"><div><strong>{p?money(p.price,p.currency):'Loading…'}</strong><span>{p?.name??'Synthetic concept device'} · test price</span></div><button className="primary" onClick={add} disabled={busy||!p||p.available===false}>{busy?'Adding…':'Add to bag'} <span>+</span></button></div><p className="fine-print">Synthetic demo product. Colour changes the illustration; it does not select a product variant.</p></div></div></section>
 <section className="editorial"><p className="eyebrow">LESS SCROLLING. MORE LIVING.</p><h2>Technology should<br/>fit into <em>your world.</em></h2><button className="text-link" onClick={()=>go('/pages/story')}>A little about our philosophy ↗</button><div className="editorial-symbol">✳</div></section></main><footer><div className="wordmark">orbit<span>®</span></div><p>A different kind of everyday.</p><span>LOCAL CONCEPT · SYNTHETIC DATA · NO PURCHASES</span></footer>
 {bag&&<div className="bag-overlay"><button className="backdrop" aria-label="Close shopping bag" onClick={()=>setBag(false)}/><aside className="bag-panel" role="dialog" aria-modal="true" aria-label="Shopping bag"><div className="bag-heading"><h2>Your bag <sup>{count}</sup></h2><button autoFocus aria-label="Close shopping bag" onClick={()=>setBag(false)}>×</button></div><p className="eyebrow">GOOD CHOICES LOOK GOOD ON YOU.</p>{count>0?<><div className="bag-item"><img src={image??undefined} alt="Synthetic device"/><div><h3>{p?.name}</h3><p>Quantity {count} · synthetic product</p><strong>{money(cart.total,cart.currency)}</strong></div></div><div className="bag-total"><span>Subtotal</span><strong>{money(cart.total,cart.currency)}</strong></div><p className="fine-print">Shipping is calculated in Baci checkout. Local preview only; do not place an order.</p><button className="primary wide" onClick={checkout} disabled={busy}>{busy?'Preparing…':'Review checkout'} <span>↗</span></button></>:<div className="empty-bag"><h3>A little room for possibility.</h3><p>Your bag is empty. Find your next everyday essential.</p><button className="primary" onClick={()=>setBag(false)}>Keep exploring ↗</button></div>}<button className="text-link" onClick={read}>Refresh bag</button><p role="status">{message}</p></aside></div>}<div className="demo-badge">LOCAL DEMO <span>•</span> Baci commerce, custom everything else</div></div>
}
